#   pip install --upgrade pip wheel "setuptools<81"   # madmom needs pkg_resources
#   pip install "Cython==3.0.1" numpy==1.26.4 scipy mido
#   pip install madmom==0.16.1 --no-build-isolation
#   pip install -r requirements.txt
#

#   - essentia         -> tried with pip as an optional extra (never

#   - Ollama + muse-glimmer:30b-mlx, the MOSS-Music code clone and the model
